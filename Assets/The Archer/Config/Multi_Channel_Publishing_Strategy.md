### 1.2 三种商业模式

| 模式 | 描述 | 收入来源 | 适用场景 |
|------|------|----------|----------|
| **UGC 平台模式** | 运营自有 UGC 游戏平台 | 分成、广告、会员 | 长期生态建设 |
| **小游戏矩阵模式** | 批量生产小游戏上架各平台 | 广告、内购 | 快速变现 |
| **独立游戏模式** | 精品游戏独立上架应用商店 | 付费下载、内购 | 海外市场 |

```csharp
// AdManager.cs - 统一广告管理

public class AdManager : MonoBehaviour
{
    public static AdManager Instance { get; private set; }
    
    private IChannelSDK channelSDK;
    
    // 广告配置
    private AdConfig adConfig;
    
    void Awake()
    {
        Instance = this;
        channelSDK = ChannelSDKFactory.Create();
        LoadAdConfig();
    }
    
    private void LoadAdConfig()
    {
        // 从服务器加载广告配置
        adConfig = new AdConfig
        {
            BannerAdUnitId = GetAdUnitId("banner"),
            InterstitialAdUnitId = GetAdUnitId("interstitial"),
            RewardedAdUnitId = GetAdUnitId("rewarded"),
            
            // 广告展示策略
            InterstitialInterval = 180, // 3分钟间隔
            MaxDailyInterstitials = 20,
            RewardedCooldown = 30
        };
    }
    
    private string GetAdUnitId(string adType)
    {
        #if UNITY_WEIXINMINIGAME
            return adType switch
            {
                "banner" => "adunit-xxxxxxxx",
                "interstitial" => "adunit-yyyyyyyy",
                "rewarded" => "adunit-zzzzzzzz",
                _ => ""
            };
        #elif UNITY_ANDROID
            return adType switch
            {
                "banner" => "ca-app-pub-xxx/banner",
                "interstitial" => "ca-app-pub-xxx/interstitial",
                "rewarded" => "ca-app-pub-xxx/rewarded",
                _ => ""
            };
        #else
            return "";
        #endif
    }
    
    // 显示激励视频广告
    public void ShowRewardedAd(string placement, Action<bool> onResult)
    {
        // 检查冷却时间
        if (!CanShowRewardedAd())
        {
            onResult?.Invoke(false);
            return;
        }
        
        // 上报广告请求
        Analytics.ReportEvent("ad_request", new Dictionary<string, object>
        {
            {"ad_type", "rewarded"},
            {"placement", placement}
        });
        
        channelSDK.ShowRewardedAd(adConfig.RewardedAdUnitId, (success) =>
        {
            // 上报广告结果
            Analytics.ReportEvent("ad_result", new Dictionary<string, object>
            {
                {"ad_type", "rewarded"},
                {"placement", placement},
                {"success", success}
            });
            
            onResult?.Invoke(success);
        });
    }
    
    // 显示插屏广告
    public void ShowInterstitialAd(string placement)
    {
        if (!CanShowInterstitialAd())
            return;
            
        channelSDK.ShowInterstitialAd(adConfig.InterstitialAdUnitId, () =>
        {
            // 广告关闭回调
        });
    }
    
    private bool CanShowRewardedAd()
    {
        // 检查冷却时间和每日上限
        return true;
    }
    
    private bool CanShowInterstitialAd()
    {
        // 检查间隔时间和每日上限
        return true;
    }
}
```
